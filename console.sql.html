CREATE TABLE Employee
(
    employee_id  varchar(8) PRIMARY KEY,
    employee_first    varchar(20),
    employee_last     varchar(20),
    employee_password varchar(20)
);

CREATE TABLE Appointment
(
    appt_number varchar(9) PRIMARY KEY,
    notes      varchar(30),
    check_in    string(10),
    check_out   string(10),
    appt_date date(2),                                                                      -- YYYY-MM-DD HH:MM
    assigned_by varchar(8),
    CONSTRAINT fkAssigned_by FOREIGN KEY (assigned_by) REFERENCES Employee(employee_id),    --Appointments must be assigned by an employee
    CONSTRAINT fkPaid FOREIGN KEY (appt_number) REFERENCES Payment(payment_id)              --Appointments must have a payment
);

CREATE TABLE Payment
(
    payment_id     varchar(9) PRIMARY KEY,
    payment_status varchar(30),
    payment_method varchar(13),
    payment_amount decimal,
    CONSTRAINT fkPayment_customer FOREIGN KEY(payment_id) REFERENCES Customer(customer_id)  --Payments must be provided by a customer
);

CREATE TABLE Receipt
(
    receipt_id   varchar(9) PRIMARY KEY,
    receipt_desc varchar(50),
    CONSTRAINT fkPayment_receipt FOREIGN KEY(receipt_id) REFERENCES Payment(payment_id)     --Receipts must be referencing a payment
);

CREATE TABLE Customer
(
    customer_id    varchar(9) PRIMARY KEY,
    customer_first varchar(20),
    customer_last  varchar(20),
    customer_phone varchar(10)
);

CREATE TABLE Pet
(
    pet_name   varchar(20),
    pet_breed  varchar(20),
    pet_health varchar(50),
    pet_sex    char(1),
    pet_owner  varchar(9),
    CONSTRAINT fkPet_owner FOREIGN KEY(pet_owner) REFERENCES Customer(customer_id)         --Pets must have an owner
);

-- Statement for dashboard >> querying the necessary information regarding the appointment
SELECT appt_number, appt_date, check_in, check_out, PET.pet_name, CUSTOMER.customer_last, CUSTOMER.customer_first, EMPLOYEE.employee_last, EMPLOYEE.employee_first, CUSTOMER.customer_phone, notes, PAYMENT.payment_id
FROM APPOINTMENT JOIN EMPLOYEE
ON assigned_by = employee_id
JOIN PAYMENT
ON appt_number = payment_id
JOIN Customer
ON payment_id = customer_id
JOIN PET
ON customer_id = pet_owner;
